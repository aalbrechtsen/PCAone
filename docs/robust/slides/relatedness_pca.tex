% ============================================================== why
\sect{Why relatives break PCA}

\begin{frame}{Relatives share alleles}
\begin{itemize}
  \item PCA looks at how genotypes co-vary between individuals.
  \item Two sources: shared \textbf{ancestry} and recent \textbf{relatedness}.
\end{itemize}
\[ E\big[(g_i-2\pi)(g_j-2\pi)\big]
 = \underbrace{4(\pi_i-\pi)(\pi_j-\pi)}_{\text{ancestry: few axes $\to$ the PCs}}
 \;+\; \underbrace{4\,\phi_{ij}\sqrt{\pi_i(1-\pi_i)\,\pi_j(1-\pi_j)}}_{\text{relatedness: should not shape the PCs}} \]
\aside{$g_i$: genotype of individual $i$ at a SNP (0, 1 or 2 copies of an allele);
$\pi$: allele frequency in the sample; $\pi_i$: individual allele frequency, given $i$'s ancestry;\\
$\phi_{ij}$: kinship given ancestry, the chance that an allele drawn from each is
identical by descent ($\tfrac14$ parent--offspring, $\tfrac18$ 2nd degree, 0 unrelated).}
\vspace{4pt}
\begin{itemize}
  \item A family is a small, tight cluster --- PCA spends components on it.
\end{itemize}
\end{frame}

\begin{frame}{The usual fixes}
\begin{itemize}
  \item \textbf{Drop relatives} --- you lose people.
  \item \textbf{Project relatives} (PC-AiR) --- they are no longer in-sample.
  \item \textbf{Screen with KING} --- biased for admixed people (later).
\end{itemize}
\take{We want to keep everyone in the PCA.}
\end{frame}

% ============================================================== small N
\sect{The small-$N$ problem}

\begin{frame}{The GRM diagonal}
\centering
\includegraphics[width=0.92\linewidth]{smalln_grm.pdf}
\begin{itemize}
  \item Diagonal $=$ ancestry part $+$ heterozygosity noise --- and the noise differs by population.
  \item PCA spends an axis on it, \textbf{even without relatives}.
\end{itemize}
\end{frame}

% ============================================================== methods
\sect{The methods}

\begin{frame}{Small $N$: fix the diagonal --- three options}
\begin{enumerate}
  \item \textbf{Impute it}: predict the diagonal from the off-diagonal entries; the rest
    goes into $S$ (\opt{impute-diag}).
  \item \textbf{Remove the heterozygosity} from it: Chen \& Storey (next slide).
  \item \textbf{Whiten}: scale each person by their own noise, so the noise is the same for everyone.
\end{enumerate}
\vspace{6pt}
\centering
\begin{tabular}{@{}lcc@{}}
\toprule
5 per population, mean min $R^2$ & no relatives & with relatives \\
\midrule
diagonal kept & 0.02 & 0.02 \\
diagonal imputed & \textbf{0.99} & \textbf{0.99} \\
\bottomrule
\end{tabular}
\end{frame}

\begin{frame}{Remove the heterozygosity: Chen \& Storey}
\[ H = \tfrac1M\,GG^\top - \operatorname{diag}(h_i), \qquad
   h_i = \tfrac1M\textstyle\sum_s g_{is}(2-g_{is}) \]
\vspace{-6pt}
\centering
\includegraphics[width=0.92\linewidth]{cs_pcs.pdf}\par
\aside{$G$ \emph{not} centred; $h_i$: individual $i$'s heterozygosity. 5 per population, no relatives.}
\take{Not centred: PC1 is the mean --- ancestry starts at PC1+1.}
\end{frame}

\begin{frame}{A trap: centring}
\begin{itemize}
  \item The GRM uses \emph{centred} genotypes.
  \item Centring spreads each person's diagonal over the whole matrix:
\end{itemize}
\[ C_{ij} \;\ni\; -\frac{d_i + d_j}{N} \]
\begin{itemize}
  \item Freeing the diagonal then frees nothing.
\end{itemize}
\end{frame}

\begin{frame}{Work uncentred}
\centering
\begin{tabular}{@{}lc@{}}
\toprule
low-rank fit, diagonal free & failures, 5 per population \\
\midrule
centred, scaled GRM & \alert{82\%} \\
uncentred, scaled GRM & \textbf{0\%} \\
\bottomrule
\end{tabular}
\take{Scaling does not matter. Centring does.}
\end{frame}

\begin{frame}{Small $N$ as $L + S$ --- robust PCA}
\centering
\includegraphics[width=\linewidth]{smalln_ls.pdf}
\take{$L$: GRM, diagonal imputed from the top PCs. $S$: the rest.}
\aside{5 per population, no relatives. With relatives, $S$ will also hold the related pairs.\\
Two ways to fit it: principal component pursuit (PCP; \texttt{aarobust-kin}) or a fixed-rank fit (the other three).\\
Uncentred matrices: PC1 is the mean, so we write PC$k$+1 for the PC that corresponds to standard PC$k$.}
\end{frame}

\begin{frame}{Imputing the diagonal (1): the top $r$ PCs}
\centering
\includegraphics[width=0.88\linewidth]{impute_topr.pdf}
\begin{enumerate}
  \item Guess the diagonal (e.g.\ the row means).
  \item Keep the top $r$ PCs: \;$L_r = V_{[\,,1:r]}\,\Lambda_{[1:r,1:r]}\,V_{[\,,1:r]}^\top$ \;(eigenvectors $V$, eigenvalues $\Lambda$).
  \item Set the diagonal to the prediction: \;$A_{ii} \leftarrow (L_r)_{ii} = \sum_{k=1}^{r} \lambda_k V_{ik}^2$.
  \item Repeat until it stops changing.
\end{enumerate}
\aside{$r$ grows while the next eigenvalue is above the noise edge. Used by the uncentred GRM fit, \texttt{detect-white} and \texttt{dwg}.}
\end{frame}

\begin{frame}{Imputing the diagonal (2): the nuclear norm}
\centering
\includegraphics[width=\linewidth]{impute_nuclear.pdf}
\[ \min_{\text{diagonal}} \; \|L\|_* = \sum_k |\lambda_k| \qquad \text{with } L_{ij} = C_{ij} \text{ for } i \ne j \]
\aside{No rank to choose; used by PCP (\texttt{aarobust-kin}). Every eigenvector depends on the diagonal, so all of them take part.}
\end{frame}

\begin{frame}{What is whitening?}
\centering
\includegraphics[width=\linewidth]{whitening.pdf}
\[ \Sigma = \operatorname{diag}(v_1,\dots,v_N), \qquad \text{whitened matrix} = \Sigma^{-1/2} A\, \Sigma^{-1/2}, \quad
   \text{entries } A_{ij}/\sqrt{v_i v_j} \]
\aside{$v_i$: person $i$'s noise (the leftover diagonal). After whitening the noise is the same for everyone, and a constant
diagonal does not change the eigenvectors. PCs: eigenvectors of the whitened matrix, multiplied back by $\Sigma^{1/2}$.
With relatives, $\Sigma$ will also hold the related pairs.}
\end{frame}

\begin{frame}{Uncentred GRM fit}
\centering
\includegraphics[width=0.85\linewidth,trim=0 0 0 0.9cm,clip]{ga_small_fit.pdf}\par
\vspace{-2pt}
{\small
\[ \text{add PC } r{+}1 \text{ if }\; \lambda_{r+1}(A - S) > 2\sigma\sqrt N, \qquad
   \sigma^2 = \frac{1}{M^2}\sum_s \frac{\operatorname{Var}(g_{is}g_{js})}{w_s^2} \]}
\vspace{-6pt}
\aside{$\sigma$: the spread of an off-diagonal entry if there were no structure (unrelated individuals, one population);
$2\sigma\sqrt N$: the largest eigenvalue of such pure noise (Wigner's semicircle).
$\operatorname{Var}(g_ig_j) = E[g^2]^2 - (2f)^4$, $E[g^2] = 2f(1-f) + 4f^2$, $w_s = 2f_s(1-f_s)$.}
\take{Diagonal free; add imputation PCs while they rise above the noise edge.}
\end{frame}

\begin{frame}{\texttt{aarobust-kin}: PCP of the GRM}
\centering
\includegraphics[width=\linewidth,trim=0 0 0 0.9cm,clip]{ga_small_aarobust.pdf}
\take{Diagonal imputed with all eigenvectors --- no rank, but slow.}
\aside{$L = C$ off the diagonal; its diagonal is the one with the smallest $\sum_k|\lambda_k|$ (nuclear norm).
$C = L + S$ exactly, $S$ takes the rest of the diagonal. Each iteration needs a full eigendecomposition.}
\end{frame}

\begin{frame}{\texttt{detect-white}: the Chen \& Storey matrix}
\centering
\includegraphics[height=0.74\textheight,trim=0 0 0 0.9cm,clip]{ga_small_dw.pdf}
\take{The same fit on $H$, then each person is scaled by their own noise.}
\end{frame}

\begin{frame}{\texttt{dwg}: uncentred GRM, then whitened and centred}
\centering
\includegraphics[height=0.74\textheight,trim=0 0 0 0.9cm,clip]{ga_small_dwg.pdf}
\take{The uncentred fit, then whitened and centred --- PC1 is ancestry.}
\end{frame}

\begin{frame}{Four ways to fit $L + S$}
\centering
\footnotesize
\begin{tabular}{@{}llll@{}}
\toprule
method & matrix & diagonal imputed from & PCs from \\
\midrule
uncentred GRM fit & scaled, uncentred GRM & top $r$ PCs & $L$ \\
\texttt{aarobust-kin} & centred, scaled GRM & all eigenvectors (min $\sum|\lambda|$) & $L$ \\
\texttt{detect-white} & Chen \& Storey (raw) & top $r$ PCs & whitened Gram \\
\texttt{dwg} & scaled, uncentred GRM & top $r$ PCs & whitened, centred \\
\bottomrule
\end{tabular}\\[3pt]
\aside{$r$: chosen by the noise edge.}
\take{All four impute the diagonal and put the rest in $S$.}
\end{frame}

\begin{frame}{Small $N$: comparing the methods}
\centering
\includegraphics[width=\linewidth]{smalln_compare.pdf}
\take{The four are equally accurate --- \texttt{aarobust-kin} is the slow one.}
\aside{No relatives, admixTjeck2, 5 replicates per $n$ (numbers in the backup). Run times: PCAone C++.}
\end{frame}

\begin{frame}{Small $N$: the suggestion}
\begin{itemize}
  \item Use the \textbf{uncentred} GRM.
  \item Leave the diagonal \textbf{free} (imputed), rather than correcting it as Chen \& Storey do:
    their correction assumes HWE within individuals.
  \item Built into \opt{robust} (\texttt{dwg}), together with the relatedness correction.
\end{itemize}
\vspace{6pt}
\centering
\begin{tabular}{@{}lcccc@{}}
\toprule
PCA wrong, per population: & 5 & 10 & 20 & 40 \\
\midrule
standard PCA & \alert{100\%} & \alert{78\%} & \alert{60\%} & 0\% \\
\texttt{dwg} & \textbf{0\%} & \textbf{0\%} & \textbf{0\%} & \textbf{0\%} \\
\bottomrule
\end{tabular}\\[3pt]
\aside{8 scenarios $\times$ 5 replicates, with and without relatives, incl.\ an inbred population and genotype errors.
The Chen \& Storey diagonal works without relatives, but fails in 80\% of runs with an inbred population.}
\end{frame}

% ============================================================== relatives
\begin{frame}\vfill\centering{\color{accent}\huge\bfseries The methods: relatives}\vfill\end{frame}

\begin{frame}{The idea}
\begin{itemize}
  \item Split the matrix: \textbf{structure} (low rank) $+$ \textbf{related pairs} (sparse).
  \item A pair is related if its kinship $> \tau = 2^{-3.5} \approx 0.088$
    \aside{(between 2nd and 3rd degree)}.
  \item The diagonal stays free.
\end{itemize}
\end{frame}

\begin{frame}{Robust PCA of the GRM (\texttt{aarobust-kin})}
\centering
\includegraphics[width=\linewidth,trim=0 0 0 1.05cm,clip]{ga_aarobust.pdf}
\take{Accurate, but slow: a full eigendecomposition every iteration.}
\end{frame}

\begin{frame}{Detect, then whiten and centre (\texttt{dwg}, the default)}
\centering
\includegraphics[height=0.74\textheight,trim=0 0 0 0.9cm,clip]{ga_dwg.pdf}
\take{Fast: only the top few eigenvectors. No HWE assumption.}
\end{frame}

\begin{frame}{How many PCs? The data decide}
\begin{itemize}
  \item Add axes while they rise above the \textbf{noise edge}.
  \item An axis resting on 2--3 people is a \textbf{family}, not ancestry.
  \item So \texttt{-k} is only an upper bound.
\end{itemize}
\vspace{6pt}
\centering
\begin{tabular}{@{}lcc@{}}
\toprule
runs that change between $k=3$ and $k=10$ & old & \texttt{dwg} \\
\midrule
of 240 & 12 & \textbf{2} \\
\bottomrule
\end{tabular}
\end{frame}

\begin{frame}{Whiten --- don't just remove the pairs}
\begin{itemize}
  \item Removing the related entries still counts each sibling fully.
  \item Whitening counts a family by its \emph{effective} size.
\end{itemize}
\vspace{6pt}
\centering
\begin{tabular}{@{}lc@{}}
\toprule
family signal in the extra PCs & median $\eta^2$ \\
\midrule
standard PCA & 0.92 \\
related pairs imputed & 0.37 \\
\textbf{whitening} & \textbf{0.21} \\
\bottomrule
\end{tabular}
\end{frame}

% ============================================================== admixed
\sect{Kinship for admixed individuals}

\begin{frame}{KING is not structure-free}
\begin{itemize}
  \item Correct only for two people from the \textbf{same unadmixed} population.
  \item Relatives of different ancestry are pulled \textbf{down}.
\end{itemize}
\vspace{6pt}
\centering
\begin{tabular}{@{}lc@{}}
\toprule
true 2nd-degree pair (kinship 0.125) & KING \\
\midrule
YRI grandparent -- $\tfrac14$-YRI grandchild & \alert{$-0.013$} \\
CHB aunt -- CHB/YRI nephew & \alert{0.017} \\
\bottomrule
\end{tabular}
\take{Both are missed by a KING screen.}
\end{frame}

\begin{frame}{Use individual allele frequencies}
\begin{itemize}
  \item Predict each genotype from the person's own ancestry: $\pi_{is}$.
  \item What is left over is relatedness.
  \item PC-Relate, PCAngsd and evalAdmix: \textbf{one estimator}.
\end{itemize}
\take{Use $K-1$ PCs. More PCs absorb the relatives.}
\end{frame}

\begin{frame}{The same pairs, with ancestry taken into account}
\centering
\begin{tabular}{@{}lcc@{}}
\toprule
true 2nd-degree pair (0.125) & KING & evalAdmix-type \\
\midrule
YRI grandparent -- $\tfrac14$-YRI grandchild & \alert{$-0.013$} & 0.085 \\
CHB aunt -- CHB/YRI nephew & \alert{0.017} & 0.089 \\
\bottomrule
\end{tabular}
\end{frame}

\begin{frame}{The \texttt{dwg} screen}
\begin{enumerate}
  \item First fit with KING.
  \item evalAdmix kinship from the \textbf{ancestry axes only}.
  \item Confirm with \textbf{IBD}: unrelated $k_0\approx1$, 2nd degree $\approx0.5$.
\end{enumerate}
\vspace{6pt}
\centering
\begin{tabular}{@{}lcc@{}}
\toprule
relatives found & KING screen & \texttt{dwg} screen \\
\midrule
5--40 per population & 86--90\% & \textbf{95--100\%} \\
\bottomrule
\end{tabular}
\end{frame}

\begin{frame}{Not just kinship: $(k_0, k_1, k_2)$}
\begin{itemize}
  \item Parent--offspring and full sibs both have kinship $\tfrac14$.
  \item $k_0$ tells them apart: 0 vs $\tfrac14$.
\end{itemize}
\vspace{6pt}
\centering
\begin{tabular}{@{}lccc@{}}
\toprule
estimated, $N=2000$ & $k_0$ & $k_1$ & $k_2$ \\
\midrule
parent--offspring & 0.00 & 1.00 & 0.00 \\
full sibs & 0.25 & 0.50 & 0.25 \\
2nd degree & 0.50 & 0.49 & 0.01 \\
\bottomrule
\end{tabular}
\end{frame}

\begin{frame}{Two details that matter}
\begin{itemize}
  \item Keep each person's own frequencies, $\pi_i \neq \pi_j$.\\
    \aside{Pooling them calls a parent--offspring pair 83\% unrelated.}
  \item Leave the pair's \textbf{whole family} out of the frequency fit.
\end{itemize}
\end{frame}

\begin{frame}{Take home}
\begin{enumerate}
  \item Relatives break PCA: a lost axis at small $N$, family axes at large $N$.
  \item At small $N$ the diagonal breaks it too: work uncentred, diagonal free.
  \item Detect the related pairs, then whiten: everyone stays in-sample.
  \item For admixed relatives use ancestry-aware kinship, not KING.
\end{enumerate}
\vspace{8pt}
\aside{Open: missing genotypes; first cousins at $\sim$5 per population.}
\end{frame}

  \item Kinship from the residual: $\hat\phi_{ij}=R_{ij}/(2\sqrt{v_iv_j})$,
    $R=A-L$, $v_i$ the fitted noise.
  \item Whitening: $\Sigma=v^{1/2}(I+2\Psi)v^{1/2}$; PCs of
    $\Sigma^{-1/2}A\Sigma^{-1/2}$, mapped back by $\Sigma^{1/2}$.
  \item Centring: $JDJ = D-(d\mathbf 1'+\mathbf 1d')/N+\bar d\,\mathbf 1\mathbf 1'/N$.
  \item IBD likelihood: IBD0 $g_i\sim\mathrm{HWE}(\pi_i)$, $g_j\sim\mathrm{HWE}(\pi_j)$;
    IBD1 one shared allele; IBD2 $g_i=g_j$.
\end{itemize}
