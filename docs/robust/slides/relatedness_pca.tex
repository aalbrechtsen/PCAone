\begin{frame}{The goal}
\begin{itemize}
  \item PCs that reflect \textbf{ancestry}, not families.
  \item \textbf{Everyone in-sample}: nothing dropped, nothing projected.
  \item From \textbf{20 to 100{,}000} individuals.
\end{itemize}
\vspace{10pt}
\begin{itemize}
  \item And estimate the relatedness itself --- also for \textbf{admixed} people.
\end{itemize}
\end{frame}

% ============================================================== why
\sect{Why relatives break PCA}

\begin{frame}{Relatives share alleles}
\begin{itemize}
  \item PCA looks at how genotypes co-vary between individuals.
  \item Relatives co-vary beyond what ancestry explains:
\end{itemize}
\[ \operatorname{Cov}(g_i,g_j) = 4\,\phi_{ij}\;\pi(1-\pi) \]
\begin{itemize}
  \item A family is a small, tight cluster --- PCA spends components on it.
\end{itemize}
\end{frame}

\begin{frame}{Large samples: a different symptom}
\centering
\begin{tabular}{@{}lcc@{}}
\toprule
$N=2000$, 30\% in families & standard PCA & robust \\
\midrule
ancestry axes correct & yes & yes \\
relatives' placement error & 0.49 & 0.12 \\
family axes among 16 extra PCs & \alert{16} & 0 \\
\bottomrule
\end{tabular}
\take{Small N: a lost axis. Large N: misplaced relatives, family axes.}
\end{frame}

\begin{frame}{The usual fixes}
\begin{itemize}
  \item \textbf{Drop relatives} --- you lose people.
  \item \textbf{Project relatives} (PC-AiR) --- they are no longer in-sample.
  \item \textbf{Screen with KING} --- biased for admixed people (later).
\end{itemize}
\take{We want to keep everyone in the PCA.}
\end{frame}

% ============================================================== small N
\sect{The small-$N$ problem}

\begin{frame}{The GRM diagonal}
\begin{itemize}
  \item Diagonal $=$ structure $+$ heterozygosity noise.
  \item At 5 per population the noise wins --- and it differs by population.
  \item PCA spends an axis on it, \textbf{even without relatives}.
\end{itemize}
\take{That was the second hook.}
\end{frame}

\begin{frame}{Fix: ignore the diagonal}
\begin{itemize}
  \item Treat the diagonal as \textbf{unobserved}.
  \item Impute it from the off-diagonal entries (\opt{impute-diag}).
\end{itemize}
\vspace{6pt}
\centering
\begin{tabular}{@{}lcc@{}}
\toprule
5 per population, mean min $R^2$ & no relatives & with relatives \\
\midrule
diagonal kept & 0.02 & 0.02 \\
diagonal imputed & \textbf{0.99} & \textbf{0.99} \\
\bottomrule
\end{tabular}
\end{frame}

\begin{frame}{A trap: centring}
\begin{itemize}
  \item The GRM uses \emph{centred} genotypes.
  \item Centring spreads each person's diagonal over the whole matrix:
\end{itemize}
\[ C_{ij} \;\ni\; -\frac{d_i + d_j}{N} \]
\begin{itemize}
  \item Freeing the diagonal then frees nothing.
\end{itemize}
\end{frame}

\begin{frame}{Work uncentred}
\centering
\begin{tabular}{@{}lc@{}}
\toprule
low-rank fit, diagonal free & failures, 5 per population \\
\midrule
centred GRM & \alert{82\%} \\
uncentred GRM & \textbf{0\%} \\
\bottomrule
\end{tabular}
\take{Scaling does not matter. Centring does.}
\end{frame}

% ============================================================== methods
\sect{The methods}

\begin{frame}{The idea}
\begin{itemize}
  \item Split the matrix: \textbf{structure} (low rank) $+$ \textbf{related pairs} (sparse).
  \item A pair is related if its kinship $> \tau = 2^{-3.5} \approx 0.088$
    \aside{(between 2nd and 3rd degree)}.
  \item The diagonal stays free.
\end{itemize}
\end{frame}

\begin{frame}{Robust PCA of the GRM (\texttt{aarobust-kin})}
\centering
\includegraphics[width=\linewidth,trim=0 0 0 1.05cm,clip]{ga_aarobust.pdf}
\take{Accurate, but slow: a full eigendecomposition every iteration.}
\end{frame}

\begin{frame}{Detect, then whiten (\texttt{dwg}, the default)}
\centering
\includegraphics[width=\linewidth,trim=0 0 0 1.05cm,clip]{ga_dwg.pdf}
\take{Fast: only the top few eigenvectors. No HWE assumption.}
\end{frame}

\begin{frame}{How many PCs? The data decide}
\begin{itemize}
  \item Add axes while they rise above the \textbf{noise edge}.
  \item An axis resting on 2--3 people is a \textbf{family}, not ancestry.
  \item So \texttt{-k} is only an upper bound.
\end{itemize}
\vspace{6pt}
\centering
\begin{tabular}{@{}lcc@{}}
\toprule
runs that change between $k=3$ and $k=10$ & old & \texttt{dwg} \\
\midrule
of 240 & 12 & \textbf{2} \\
\bottomrule
\end{tabular}
\end{frame}

\begin{frame}{Whiten --- don't just remove the pairs}
\begin{itemize}
  \item Removing the related entries still counts each sibling fully.
  \item Whitening counts a family by its \emph{effective} size.
\end{itemize}
\vspace{6pt}
\centering
\begin{tabular}{@{}lc@{}}
\toprule
family signal in the extra PCs & median $\eta^2$ \\
\midrule
standard PCA & 0.92 \\
related pairs imputed & 0.37 \\
\textbf{whitening} & \textbf{0.21} \\
\bottomrule
\end{tabular}
\end{frame}

% ============================================================== admixed
\sect{Kinship for admixed individuals}

\begin{frame}{KING is not structure-free}
\begin{itemize}
  \item Correct only for two people from the \textbf{same unadmixed} population.
  \item Relatives of different ancestry are pulled \textbf{down}.
\end{itemize}
\vspace{6pt}
\centering
\begin{tabular}{@{}lc@{}}
\toprule
true 2nd-degree pair (kinship 0.125) & KING \\
\midrule
YRI grandparent -- $\tfrac14$-YRI grandchild & \alert{$-0.013$} \\
CHB aunt -- CHB/YRI nephew & \alert{0.017} \\
\bottomrule
\end{tabular}
\take{Both are missed by a KING screen.}
\end{frame}

\begin{frame}{Use individual allele frequencies}
\begin{itemize}
  \item Predict each genotype from the person's own ancestry: $\pi_{is}$.
  \item What is left over is relatedness.
  \item PC-Relate, PCAngsd and evalAdmix: \textbf{one estimator}.
\end{itemize}
\take{Use $K-1$ PCs. More PCs absorb the relatives.}
\end{frame}

\begin{frame}{The same pairs, with ancestry taken into account}
\centering
\begin{tabular}{@{}lcc@{}}
\toprule
true 2nd-degree pair (0.125) & KING & evalAdmix-type \\
\midrule
YRI grandparent -- $\tfrac14$-YRI grandchild & \alert{$-0.013$} & 0.085 \\
CHB aunt -- CHB/YRI nephew & \alert{0.017} & 0.089 \\
\bottomrule
\end{tabular}
\end{frame}

\begin{frame}{The \texttt{dwg} screen}
\begin{enumerate}
  \item First fit with KING.
  \item evalAdmix kinship from the \textbf{ancestry axes only}.
  \item Confirm with \textbf{IBD}: unrelated $k_0\approx1$, 2nd degree $\approx0.5$.
\end{enumerate}
\vspace{6pt}
\centering
\begin{tabular}{@{}lcc@{}}
\toprule
relatives found & KING screen & \texttt{dwg} screen \\
\midrule
5--40 per population & 86--90\% & \textbf{95--100\%} \\
\bottomrule
\end{tabular}
\end{frame}

\begin{frame}{Not just kinship: $(k_0, k_1, k_2)$}
\begin{itemize}
  \item Parent--offspring and full sibs both have kinship $\tfrac14$.
  \item $k_0$ tells them apart: 0 vs $\tfrac14$.
\end{itemize}
\vspace{6pt}
\centering
\begin{tabular}{@{}lccc@{}}
\toprule
estimated, $N=2000$ & $k_0$ & $k_1$ & $k_2$ \\
\midrule
parent--offspring & 0.00 & 1.00 & 0.00 \\
full sibs & 0.25 & 0.50 & 0.25 \\
2nd degree & 0.50 & 0.49 & 0.01 \\
\bottomrule
\end{tabular}
\end{frame}

\begin{frame}{Two details that matter}
\begin{itemize}
  \item Keep each person's own frequencies, $\pi_i \neq \pi_j$.\\
    \aside{Pooling them calls a parent--offspring pair 83\% unrelated.}
  \item Leave the pair's \textbf{whole family} out of the frequency fit.
\end{itemize}
\end{frame}

% ============================================================== scale
\sect{Scaling up}

\begin{frame}{Large $N$: no $N\times N$ matrix}
\centering
\begin{tabular}{@{}lcc@{}}
\toprule
16 threads, in memory & standard PCA & \texttt{dwg} \\
\midrule
$N=5000$ & 3.5\,s & 4.7\,s \\
$N=20{,}000$ & 24\,s & 26.5\,s \\
\bottomrule
\end{tabular}
\take{Robust PCA at about the cost of standard PCA.}
\end{frame}

\begin{frame}{Very large $N$: find relatives with a sketch}
\begin{itemize}
  \item Comparing all pairs is $N^2$ --- too slow at biobank scale.
  \item Sketch each genome; check only each person's nearest neighbours.
\end{itemize}
\vspace{6pt}
\centering
\begin{tabular}{@{}lcc@{}}
\toprule
$N=100{,}000$ & all pairs & sketch \\
\midrule
time to find relatives & 162\,s & \textbf{52\,s} \\
related pairs found & 27{,}880 & \textbf{27{,}880} \\
\bottomrule
\end{tabular}
\end{frame}

\begin{frame}{Genotype likelihoods (\texttt{-G})}
\begin{itemize}
  \item Low depth shrinks genotypes towards the mean.
  \item Undo the shrinkage; the extra noise lands on the free diagonal.
\end{itemize}
\vspace{6pt}
\centering
\begin{tabular}{@{}lcc@{}}
\toprule
PCA wrong at 4$\times$ & 10 per pop. & 5 per pop. \\
\midrule
standard PCA on likelihoods & 100\% & 100\% \\
\texttt{dwg} & \textbf{0\%} & 12.5\% \\
\bottomrule
\end{tabular}
\end{frame}

\end{verbatim}

  \item \texttt{.eigvecs}: the robust PCs.
  \item \texttt{.relpairs}: related pairs with kinship and $(k_0,k_1,k_2)$.
\end{itemize}
\vspace{6pt}
\aside{github.com/aalbrechtsen/PCAone, branch \texttt{robust-relatives}, docs/robust/}
\end{frame}

\begin{frame}{Take home}
\begin{enumerate}
  \item Relatives break PCA: a lost axis at small $N$, family axes at large $N$.
  \item At small $N$ the diagonal breaks it too: work uncentred, diagonal free.
  \item Detect the related pairs, then whiten: everyone stays in-sample.
  \item For admixed relatives use ancestry-aware kinship, not KING.
\end{enumerate}
\vspace{8pt}
\aside{Open: missing genotypes; first cousins at $\sim$5 per population.}
\end{frame}

\begin{frame}{Backup: formulas}
\small
\begin{itemize}
  \item Kinship from the residual: $\hat\phi_{ij}=R_{ij}/(2\sqrt{v_iv_j})$,
    $R=A-L$, $v_i$ the fitted noise.
  \item Whitening: $\Sigma=v^{1/2}(I+2\Psi)v^{1/2}$; PCs of
    $\Sigma^{-1/2}A\Sigma^{-1/2}$, mapped back by $\Sigma^{1/2}$.
  \item Centring: $JDJ = D-(d\mathbf 1'+\mathbf 1d')/N+\bar d\,\mathbf 1\mathbf 1'/N$.
  \item IBD likelihood: IBD0 $g_i\sim\mathrm{HWE}(\pi_i)$, $g_j\sim\mathrm{HWE}(\pi_j)$;
    IBD1 one shared allele; IBD2 $g_i=g_j$.
\end{itemize}
\end{frame}
